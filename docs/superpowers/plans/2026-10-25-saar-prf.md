# SAAR PRF Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construir a aplicação SAAR PRF (API REST Spring Boot + PostgreSQL no Neon + front HTML/CSS/TypeScript) que importa, consulta, altera, exclui, filtra e grafica os acidentes da PRF.

**Architecture:** Camadas model / repository / service / controller em `br.edu.fatec.saarprf`. A importação lê o CSV oficial em fluxo, converte cada linha (`ConversorLinhaCsv`), valida com Bean Validation, deduplica por `idPrf` e grava em lotes. O front é uma página única em TypeScript compilado para `static/js`, consumindo a API com `fetch()` e desenhando dois gráficos com Chart.js.

**Tech Stack:** Java 21, Spring Boot 4.1.2 (webmvc, data-jpa, validation), PostgreSQL (Neon), Apache Commons CSV 1.14.1, JUnit 5 + Mockito + MockMvc, TypeScript 5.9.3, Chart.js 4.4.1.

**Spec:** `docs/superpowers/specs/2026-10-25-saar-prf-design.md`

```java
package br.edu.fatec.saarprf.service;

import java.util.Locale;

public final class Normalizacao {

    private Normalizacao() {
    }

    public static String texto(String valor) {
        if (valor == null) {
            return null;
        }
        String limpo = valor.trim().replaceAll("\\s+", " ");
        return limpo.isEmpty() ? null : limpo;
    }

    public static String maiusculo(String valor) {
        String limpo = texto(valor);
        return limpo == null ? null : limpo.toUpperCase(Locale.ROOT);
    }
}
```

```java
package br.edu.fatec.saarprf.service;

import br.edu.fatec.saarprf.model.Acidente;
import org.springframework.stereotype.Component;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.LocalTime;
import java.time.format.DateTimeFormatter;
import java.time.format.DateTimeParseException;
import java.util.Map;

@Component
public class ConversorLinhaCsv {

    private static final DateTimeFormatter DATA_BRASILEIRA = DateTimeFormatter.ofPattern("dd/MM/yyyy");

    public Acidente converter(Map<String, String> campos) {
        Acidente a = new Acidente();
        a.setIdPrf(converterIdPrf(valor(campos, "id")));
        a.setDataOcorrencia(converterData(valor(campos, "data_inversa")));
        a.setHorario(converterHorario(valor(campos, "horario")));
        a.setUf(Normalizacao.maiusculo(valor(campos, "uf")));
        a.setRodovia(converterRodovia(valor(campos, "br")));
        a.setQuilometro(converterDecimal(valor(campos, "km"), "quilômetro inválido"));
        a.setMunicipio(Normalizacao.maiusculo(valor(campos, "municipio")));
        a.setCausaAcidente(valor(campos, "causa_acidente"));
        a.setTipoAcidente(valor(campos, "tipo_acidente"));
        a.setClassificacao(converterClassificacao(valor(campos, "classificacao_acidente")));
        a.setMortos(converterInteiro(valor(campos, "mortos"), "número de mortos inválido"));
        a.setFeridos(converterInteiro(valor(campos, "feridos"), "número de feridos inválido"));
        a.setLatitude(converterDecimal(valor(campos, "latitude"), "latitude inválida"));
        a.setLongitude(converterDecimal(valor(campos, "longitude"), "longitude inválida"));
        return a;
    }

    // a PRF usa o texto "NA" para marcar valor ausente
    private String valor(Map<String, String> campos, String coluna) {
        String valor = Normalizacao.texto(campos.get(coluna));
        return valor == null || valor.equalsIgnoreCase("NA") ? null : valor;
    }

    // ids corrompidos (ex.: "8e+05", notação científica gerada por planilha) não são recuperáveis
    private Long converterIdPrf(String valor) {
        if (valor == null || !valor.matches("\\d+")) {
            throw new LinhaInvalidaException("id PRF inválido");
        }
        return Long.valueOf(valor);
    }

    // o arquivo usa aaaa-mm-dd, mas o dicionário oficial documenta dd/mm/aaaa: aceitamos os dois
    private LocalDate converterData(String valor) {
        if (valor == null) {
            throw new LinhaInvalidaException("data ausente");
        }
        try {
            return valor.contains("/") ? LocalDate.parse(valor, DATA_BRASILEIRA) : LocalDate.parse(valor);
        } catch (DateTimeParseException e) {
            throw new LinhaInvalidaException("data inválida");
        }
    }

    private LocalTime converterHorario(String valor) {
        if (valor == null) {
            throw new LinhaInvalidaException("horário ausente");
        }
        try {
            return LocalTime.parse(valor);
        } catch (DateTimeParseException e) {
            throw new LinhaInvalidaException("horário inválido");
        }
    }

    // br = 0 é o marcador da PRF para rodovia não identificada
    private String converterRodovia(String valor) {
        if (valor == null) {
            throw new LinhaInvalidaException("rodovia não identificada");
        }
        if (!valor.matches("\\d+")) {
            throw new LinhaInvalidaException("rodovia inválida");
        }
        int numero = Integer.parseInt(valor);
        if (numero == 0) {
            throw new LinhaInvalidaException("rodovia não identificada");
        }
        return String.format("BR-%03d", numero);
    }

    private String converterClassificacao(String valor) {
        if (valor == null) {
            throw new LinhaInvalidaException("classificação ausente");
        }
        return valor;
    }

    // o arquivo usa vírgula decimal, o dicionário documenta ponto: aceitamos os dois
    private BigDecimal converterDecimal(String valor, String motivo) {
        if (valor == null) {
            throw new LinhaInvalidaException(motivo);
        }
        try {
            return new BigDecimal(valor.replace(',', '.'));
        } catch (NumberFormatException e) {
            throw new LinhaInvalidaException(motivo);
        }
    }

    private Integer converterInteiro(String valor, String motivo) {
        if (valor == null) {
            throw new LinhaInvalidaException(motivo);
        }
        try {
            return Integer.valueOf(valor);
        } catch (NumberFormatException e) {
            throw new LinhaInvalidaException(motivo);
        }
    }
}
```

```java
    String FILTRO = """

                           @Param("dataInicio") LocalDate dataInicio,

                                       @Param("classificacao") String classificacao,
                                       @Param("dataInicio") LocalDate dataInicio,
                                       @Param("dataFim") LocalDate dataFim,
```

```java
package br.edu.fatec.saarprf.service;

import br.edu.fatec.saarprf.dto.ContagemCausa;
import br.edu.fatec.saarprf.dto.ContagemMensal;
import br.edu.fatec.saarprf.dto.FiltroAcidente;

import br.edu.fatec.saarprf.model.Acidente;
import br.edu.fatec.saarprf.repository.AcidenteRepository;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Pageable;
import org.springframework.data.domain.Sort;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;
import java.util.Optional;

@Service
public class AcidenteService {

    private static final int TAMANHO_MAXIMO_PAGINA = 200;
    private static final int QUANTIDADE_TOP_CAUSAS = 10;
    private static final Sort ORDEM_PADRAO = Sort.by(
            Sort.Order.desc("dataOcorrencia"), Sort.Order.desc("horario"), Sort.Order.desc("id"));

    private final AcidenteRepository repository;

    public AcidenteService(AcidenteRepository repository) {
        this.repository = repository;
    }

    public PaginaResposta<Acidente> listar(int pagina, int tamanho) {
        return PaginaResposta.de(repository.findAll(paginacao(pagina, tamanho)));
    }

    public Optional<Acidente> buscarPorId(Long id) {
        return repository.findById(id);
    }

    @Transactional
    public Acidente criar(Acidente acidente) {
        acidente.setId(null);
        // registros cadastrados manualmente não têm identificador de origem na PRF
        acidente.setIdPrf(null);
        normalizar(acidente);
        return repository.save(acidente);
    }

    @Transactional
    public Optional<Acidente> atualizar(Long id, Acidente dados) {
        return repository.findById(id).map(existente -> {
            copiarDados(dados, existente);
            normalizar(existente);
            return repository.save(existente);
        });
    }

    @Transactional
    public boolean excluir(Long id) {
        if (!repository.existsById(id)) {
            return false;
        }
        repository.deleteById(id);
        return true;
    }

    public PaginaResposta<Acidente> filtrar(FiltroAcidente filtro, int pagina, int tamanho) {
        return PaginaResposta.de(repository.filtrar(filtro.uf(), filtro.classificacao(),
                filtro.dataInicio(), filtro.dataFim(), paginacao(pagina, tamanho)));
    }

    public List<ContagemCausa> topCausas(FiltroAcidente filtro) {
        return repository.contarPorCausa(filtro.uf(), filtro.classificacao(),
                filtro.dataInicio(), filtro.dataFim(), PageRequest.of(0, QUANTIDADE_TOP_CAUSAS));
    }

    public List<ContagemMensal> evolucaoMensal(FiltroAcidente filtro) {
        return repository.contarPorMes(filtro.uf(), filtro.classificacao(),
                filtro.dataInicio(), filtro.dataFim());
    }

    private Pageable paginacao(int pagina, int tamanho) {
        return PageRequest.of(Math.max(pagina, 0), Math.clamp(tamanho, 1, TAMANHO_MAXIMO_PAGINA), ORDEM_PADRAO);
    }

    private void normalizar(Acidente acidente) {
        acidente.setUf(Normalizacao.maiusculo(acidente.getUf()));
        acidente.setMunicipio(Normalizacao.maiusculo(acidente.getMunicipio()));
        acidente.setCausaAcidente(Normalizacao.texto(acidente.getCausaAcidente()));
        acidente.setTipoAcidente(Normalizacao.texto(acidente.getTipoAcidente()));
    }

    private void copiarDados(Acidente origem, Acidente destino) {
        destino.setDataOcorrencia(origem.getDataOcorrencia());
        destino.setHorario(origem.getHorario());
        destino.setUf(origem.getUf());
        destino.setMunicipio(origem.getMunicipio());
        destino.setRodovia(origem.getRodovia());
        destino.setQuilometro(origem.getQuilometro());
        destino.setCausaAcidente(origem.getCausaAcidente());
        destino.setTipoAcidente(origem.getTipoAcidente());
        destino.setClassificacao(origem.getClassificacao());
```

```java
package br.edu.fatec.saarprf.service;

import br.edu.fatec.saarprf.dto.RejeicaoLinha;
import br.edu.fatec.saarprf.dto.ResultadoImportacao;
import br.edu.fatec.saarprf.exception.ImportacaoException;
import br.edu.fatec.saarprf.model.Acidente;
import br.edu.fatec.saarprf.repository.AcidenteRepository;
import jakarta.persistence.EntityManager;
import jakarta.validation.ConstraintViolation;
import jakarta.validation.Validator;
import org.apache.commons.csv.CSVFormat;
import org.apache.commons.csv.CSVRecord;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.multipart.MultipartFile;

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.UncheckedIOException;
import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashSet;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Locale;
import java.util.Map;
import java.util.Set;

@Service
public class ImportacaoService {

    static final List<String> COLUNAS_OBRIGATORIAS = List.of(
            "id", "data_inversa", "horario", "uf", "br", "km", "municipio", "causa_acidente",
            "tipo_acidente", "classificacao_acidente", "mortos", "feridos", "latitude", "longitude");

    private static final int TAMANHO_LOTE = 1000;
    private static final int MAXIMO_EXEMPLOS = 20;

    private final AcidenteRepository repository;
    private final ConversorLinhaCsv conversor;
    private final Validator validator;
    private final EntityManager entityManager;

    public ImportacaoService(AcidenteRepository repository, ConversorLinhaCsv conversor,
                             Validator validator, EntityManager entityManager) {
        this.repository = repository;
        this.conversor = conversor;
        this.validator = validator;
        this.entityManager = entityManager;
    }

    @Transactional
    public ResultadoImportacao importar(MultipartFile arquivo) {
        long inicio = System.currentTimeMillis();
        validarArquivo(arquivo);

        // os arquivos oficiais da PRF são publicados em ISO-8859-1, não em UTF-8
        try (BufferedReader leitor = new BufferedReader(
                new InputStreamReader(arquivo.getInputStream(), StandardCharsets.ISO_8859_1))) {

            String[] colunas = lerCabecalho(leitor);
            CSVFormat formato = CSVFormat.Builder.create()
                    .setDelimiter(';')
                    .setQuote('"')
                    .setHeader(colunas)
                    .get();

            // ids PRF já gravados + os vistos neste arquivo: reimportar o mesmo arquivo não duplica registros
            Set<Long> idsVistos = new HashSet<>(repository.buscarIdsPrf());
            Map<String, Integer> motivos = new LinkedHashMap<>();
            List<RejeicaoLinha> exemplos = new ArrayList<>();
            List<Acidente> lote = new ArrayList<>();
            int lidas = 0;
            int importadas = 0;
            int rejeitadas = 0;
            int duplicadas = 0;

            for (CSVRecord registro : formato.parse(leitor)) {
                lidas++;
                long numeroLinha = registro.getRecordNumber() + 1;
                try {
                    Acidente acidente = converterEValidar(registro);
                    if (!idsVistos.add(acidente.getIdPrf())) {
                        duplicadas++;
                        continue;
                    }
                    lote.add(acidente);
                    importadas++;
                    if (lote.size() == TAMANHO_LOTE) {
                        gravar(lote);
                    }
                } catch (LinhaInvalidaException e) {
                    // linha inválida não interrompe a importação: é contabilizada com o motivo
                    rejeitadas++;
                    motivos.merge(e.getMessage(), 1, Integer::sum);
                    if (exemplos.size() < MAXIMO_EXEMPLOS) {
                        exemplos.add(new RejeicaoLinha(numeroLinha, e.getMessage()));
                    }
                }
            }
            gravar(lote);

            if (lidas == 0) {
                throw new ImportacaoException("O arquivo não contém registros após o cabeçalho.");
            }

            String mensagem = importadas > 0
                    ? "Importação concluída: %d registros importados.".formatted(importadas)
                    : "Nenhum registro novo importado (%d duplicados, %d rejeitados).".formatted(duplicadas, rejeitadas);
            return new ResultadoImportacao(arquivo.getOriginalFilename(), lidas, importadas, rejeitadas,
                    duplicadas, motivos, exemplos, System.currentTimeMillis() - inicio, mensagem);

        } catch (IOException | UncheckedIOException e) {
            throw new ImportacaoException("Não foi possível ler o arquivo: " + e.getMessage());
        }
    }

    private void validarArquivo(MultipartFile arquivo) {
        if (arquivo == null || arquivo.isEmpty()) {
            throw new ImportacaoException("Arquivo vazio: selecione um arquivo CSV com dados.");
        }
        String nome = arquivo.getOriginalFilename();
        if (nome == null || !nome.toLowerCase(Locale.ROOT).endsWith(".csv")) {
            throw new ImportacaoException("Extensão inválida: envie um arquivo .csv.");
        }
    }

    private String[] lerCabecalho(BufferedReader leitor) throws IOException {
        String linha = leitor.readLine();
        if (linha == null || linha.isBlank()) {
            throw new ImportacaoException("Cabeçalho ausente ou incompleto: a primeira linha está vazia.");
        }
        linha = linha.replace("\uFEFF", "");
        if (!linha.contains(";")) {
            throw new ImportacaoException("Separador incorreto: o arquivo deve usar ponto e vírgula (;) entre as colunas.");
        }
        String[] colunas = Arrays.stream(linha.split(";"))
                .map(coluna -> coluna.replace("\"", "").trim())
                .toArray(String[]::new);
        List<String> presentes = Arrays.asList(colunas);
        List<String> faltando = COLUNAS_OBRIGATORIAS.stream().filter(c -> !presentes.contains(c)).toList();
        if (!faltando.isEmpty()) {
            throw new ImportacaoException("Cabeçalho ausente ou incompleto: faltam as colunas "
                    + String.join(", ", faltando) + ".");
        }
        return colunas;
    }

    private Acidente converterEValidar(CSVRecord registro) {
        if (!registro.isConsistent()) {
            throw new LinhaInvalidaException("número de colunas incorreto");
        }
        Acidente acidente = conversor.converter(registro.toMap());
        validator.validate(acidente).stream()
                .map(ConstraintViolation::getMessage)
                .sorted()
                .findFirst()
                .ifPresent(motivo -> {
                    throw new LinhaInvalidaException(motivo);
                });
        return acidente;
    }

    // flush + clear a cada lote mantém a memória estável mesmo com 70 mil registros
    private void gravar(List<Acidente> lote) {
        if (lote.isEmpty()) {
            return;
        }
        repository.saveAll(List.copyOf(lote));
        entityManager.flush();
        entityManager.clear();
        lote.clear();
    }
}
```

```ts
import type {
  Acidente, AcidenteEntrada, ContagemCausa, ContagemMensal, ErroResposta, Filtro, PaginaResposta,
  ResultadoImportacao,
} from "./tipos.js";

const BASE = "/api/acidentes";

export class ApiErro extends Error {
  constructor(
    message: string,
    public readonly status: number,
    public readonly erros: Record<string, string> = {},
  ) {
    super(message);
  }
}

async function requisitar<T>(url: string, opcoes: RequestInit = {}): Promise<T> {
  const resposta = await fetch(url, opcoes);
  if (!resposta.ok) {
    const corpo = (await resposta.json().catch(() => null)) as ErroResposta | null;
    throw new ApiErro(corpo?.mensagem ?? `Erro HTTP ${resposta.status}`, resposta.status, corpo?.erros ?? {});
  }
  if (resposta.status === 204) {
    return undefined as T;
  }
  return (await resposta.json()) as T;
}

function json(method: string, corpo: unknown): RequestInit {
  return { method, headers: { "Content-Type": "application/json" }, body: JSON.stringify(corpo) };
}

function parametros(filtro: Filtro, extras: Record<string, number> = {}): string {
  const busca = new URLSearchParams();
  for (const [chave, valor] of Object.entries({ ...filtro, ...extras })) {
    if (valor !== "") busca.set(chave, String(valor));
  }
  return busca.toString();
}

export const api = {
  listar: (pagina: number, tamanho: number) =>
    requisitar<PaginaResposta<Acidente>>(`${BASE}?page=${pagina}&size=${tamanho}`),

  filtrar: (filtro: Filtro, pagina: number, tamanho: number) =>
    requisitar<PaginaResposta<Acidente>>(`${BASE}/filtro?${parametros(filtro, { page: pagina, size: tamanho })}`),

  async buscar(id: number): Promise<Acidente | null> {
    try {
      return await requisitar<Acidente>(`${BASE}/${id}`);
    } catch (erro) {
      if (erro instanceof ApiErro && erro.status === 404) return null;
      throw erro;
    }
  },

  criar: (dados: AcidenteEntrada) => requisitar<Acidente>(BASE, json("POST", dados)),

  atualizar: (id: number, dados: AcidenteEntrada) => requisitar<Acidente>(`${BASE}/${id}`, json("PUT", dados)),

  excluir: (id: number) => requisitar<void>(`${BASE}/${id}`, { method: "DELETE" }),

  importar(arquivo: File): Promise<ResultadoImportacao> {
    const dados = new FormData();
    dados.append("arquivo", arquivo);
    return requisitar<ResultadoImportacao>(`${BASE}/importar`, { method: "POST", body: dados });
  },

  topCausas: (filtro: Filtro) => requisitar<ContagemCausa[]>(`${BASE}/estatisticas/causas?${parametros(filtro)}`),

  mensal: (filtro: Filtro) => requisitar<ContagemMensal[]>(`${BASE}/estatisticas/mensal?${parametros(filtro)}`),
};
```

```ts
import { el, elemento } from "./dom.js";
import { formatarData, formatarNumero } from "./formato.js";
import type { Acidente, PaginaResposta } from "./tipos.js";

export interface AcoesTabela {
  editar(acidente: Acidente): void;
  excluir(acidente: Acidente): void;
}

const TOTAL_COLUNAS = 10;

function botao(texto: string, acao: () => void, classe: string): HTMLButtonElement {
  const b = elemento("button", texto, `pequeno ${classe}`);
  b.type = "button";
  b.addEventListener("click", acao);
  return b;
}

export function renderizarTabela(pagina: PaginaResposta<Acidente>, acoes: AcoesTabela): void {
  const corpo = el<HTMLTableSectionElement>("corpo-tabela");
  corpo.replaceChildren();

  if (pagina.conteudo.length === 0) {
    const celula = corpo.insertRow().insertCell();
    celula.colSpan = TOTAL_COLUNAS;
    celula.className = "vazio";
    celula.textContent = "Nenhum registro encontrado.";
  }

  for (const a of pagina.conteudo) {
    const linha = corpo.insertRow();
    const valores = [
      String(a.id),
      formatarData(a.dataOcorrencia),
      a.horario.slice(0, 5),
      a.uf,
      a.municipio,
      `${a.rodovia} km ${formatarNumero(a.quilometro)}`,
      a.causaAcidente,
      a.classificacao,
      `${a.mortos} / ${a.feridos}`,
    ];
    for (const valor of valores) linha.insertCell().textContent = valor;
    linha.insertCell().append(
      botao("Editar", () => acoes.editar(a), "secundario"),
      botao("Excluir", () => acoes.excluir(a), "perigo"),
    );
  }

  el("pag-info").textContent = pagina.totalPaginas === 0
    ? "Sem páginas"
    : `Página ${pagina.pagina + 1} de ${formatarNumero(pagina.totalPaginas)}`;
  el<HTMLButtonElement>("pag-anterior").disabled = pagina.pagina === 0;
  el<HTMLButtonElement>("pag-proxima").disabled = pagina.pagina + 1 >= pagina.totalPaginas;
}
```

```ts
import { el } from "./dom.js";
import type { Acidente, AcidenteEntrada } from "./tipos.js";

type CampoFormulario = keyof AcidenteEntrada;

const CAMPOS: CampoFormulario[] = [
  "dataOcorrencia", "horario", "uf", "municipio", "rodovia", "quilometro", "causaAcidente",
  "tipoAcidente", "classificacao", "mortos", "feridos", "latitude", "longitude",
];

function campo(nome: CampoFormulario): HTMLInputElement | HTMLSelectElement {
  return el<HTMLInputElement | HTMLSelectElement>(`campo-${nome}`);
}

function textoOuNulo(nome: CampoFormulario): string | null {
  const valor = campo(nome).value.trim();
  return valor === "" ? null : valor;
}

function numeroOuNulo(nome: CampoFormulario): number | null {
  const valor = campo(nome).value.trim();
  return valor === "" ? null : Number(valor);
}

function limparErros(): void {
  for (const nome of CAMPOS) {
    const c = campo(nome);
    c.classList.remove("invalido");
    c.title = "";
  }
}

export function preencherFormulario(a: Acidente): void {
  limparErros();
  el<HTMLInputElement>("campo-id").value = String(a.id);
  campo("dataOcorrencia").value = a.dataOcorrencia;
  campo("horario").value = a.horario.slice(0, 5);
  campo("uf").value = a.uf;
  campo("municipio").value = a.municipio;
  campo("rodovia").value = String(Number(a.rodovia.replace("BR-", "")));
  campo("quilometro").value = String(a.quilometro);
  campo("causaAcidente").value = a.causaAcidente;
  campo("tipoAcidente").value = a.tipoAcidente;
  campo("classificacao").value = a.classificacao;
  campo("mortos").value = String(a.mortos);
  campo("feridos").value = String(a.feridos);
  campo("latitude").value = String(a.latitude);
  campo("longitude").value = String(a.longitude);
  el("titulo-form").textContent = `Editar registro #${a.id}`;
  el("secao-formulario").scrollIntoView({ behavior: "smooth" });
}

export function lerFormulario(): { id: number | null; dados: AcidenteEntrada } {
  const id = el<HTMLInputElement>("campo-id").value;
  const rodovia = campo("rodovia").value.trim();
  return {
    id: id ? Number(id) : null,
    dados: {
      dataOcorrencia: textoOuNulo("dataOcorrencia"),
      horario: textoOuNulo("horario"),
      uf: campo("uf").value,
      municipio: campo("municipio").value.trim(),
      rodovia: rodovia ? `BR-${rodovia.padStart(3, "0")}` : "",
      quilometro: numeroOuNulo("quilometro"),
      causaAcidente: campo("causaAcidente").value.trim(),
      tipoAcidente: campo("tipoAcidente").value.trim(),
      classificacao: campo("classificacao").value,
      mortos: numeroOuNulo("mortos"),
      feridos: numeroOuNulo("feridos"),
      latitude: numeroOuNulo("latitude"),
      longitude: numeroOuNulo("longitude"),
    },
  };
}

export function limparFormulario(): void {
  el<HTMLFormElement>("form-acidente").reset();
  el<HTMLInputElement>("campo-id").value = "";
  el("titulo-form").textContent = "Cadastrar acidente";
  limparErros();
}

export function marcarErros(erros: Record<string, string>): void {
  limparErros();
  for (const [nome, mensagem] of Object.entries(erros)) {
    if (!(CAMPOS as string[]).includes(nome)) continue;
    const c = campo(nome as CampoFormulario);
    c.classList.add("invalido");
    c.title = mensagem;
  }
}
```

```ts
import { api, ApiErro } from "./api.js";
import { CLASSIFICACOES, TIPOS_ACIDENTE, UFS } from "./constantes.js";
import { renderizarDetalhe } from "./detalhe.js";
import { el, preencherOpcoes } from "./dom.js";
import { formatarNumero } from "./formato.js";
import { lerFormulario, limparFormulario, marcarErros, preencherFormulario } from "./formulario.js";
import { desenharCausas, desenharMensal } from "./graficos.js";
import { mostrarErroImportacao, renderizarResultado } from "./importacao.js";
import { mensagemDeErro, mostrarMensagem } from "./mensagens.js";
import { renderizarTabela } from "./tabela.js";
import type { Acidente, Filtro } from "./tipos.js";

const TAMANHO_PAGINA = 50;
const FILTRO_VAZIO: Filtro = { uf: "", classificacao: "", dataInicio: "", dataFim: "" };

let pagina = 0;
let filtro: Filtro = { ...FILTRO_VAZIO };

const filtroAtivo = (): boolean => Object.values(filtro).some((valor) => valor !== "");

async function carregarTabela(): Promise<void> {
  try {
    const resultado = filtroAtivo()
      ? await api.filtrar(filtro, pagina, TAMANHO_PAGINA)
      : await api.listar(pagina, TAMANHO_PAGINA);
    renderizarTabela(resultado, { editar: preencherFormulario, excluir });
    const rotulo = filtroAtivo() ? "registros no filtro" : "registros";
    el("contador").textContent = `${formatarNumero(resultado.totalElementos)} ${rotulo}`;
  } catch (erro) {
    mostrarMensagem(mensagemDeErro(erro), "erro");
  }
}

async function carregarGraficos(): Promise<void> {
  try {
    const [causas, mensal] = await Promise.all([api.topCausas(filtro), api.mensal(filtro)]);
    desenharCausas(causas);
    desenharMensal(mensal);
    if (!filtroAtivo()) {
      const sugestoes = el<HTMLDataListElement>("lista-causas");
      sugestoes.replaceChildren();
      preencherOpcoes(sugestoes, causas.map((c) => c.causa));
    }
  } catch (erro) {
    mostrarMensagem(mensagemDeErro(erro), "erro");
  }
}

const atualizarTudo = () => Promise.all([carregarTabela(), carregarGraficos()]);

async function salvar(evento: SubmitEvent): Promise<void> {
  evento.preventDefault();
  const { id, dados } = lerFormulario();
  try {
    if (id) {
      await api.atualizar(id, dados);
      mostrarMensagem(`Registro #${id} atualizado.`);
    } else {
      const criado = await api.criar(dados);
      mostrarMensagem(`Registro #${criado.id} cadastrado.`);
    }
    limparFormulario();
    await atualizarTudo();
  } catch (erro) {
    if (erro instanceof ApiErro) marcarErros(erro.erros);
    mostrarMensagem(mensagemDeErro(erro), "erro");
  }
}

async function excluir(a: Acidente): Promise<void> {
  if (!confirm(`Excluir o registro #${a.id} (${a.municipio}/${a.uf})?`)) return;
  try {
    await api.excluir(a.id);
    mostrarMensagem(`Registro #${a.id} excluído.`);
    renderizarDetalhe(null, preencherFormulario);
    await atualizarTudo();
  } catch (erro) {
    mostrarMensagem(mensagemDeErro(erro), "erro");
  }
}

async function buscarPorId(): Promise<void> {
  const id = Number(el<HTMLInputElement>("busca-id").value);
  if (!Number.isInteger(id) || id <= 0) {
    mostrarMensagem("Informe um ID válido.", "erro");
    return;
  }
  try {
    const acidente = await api.buscar(id);
    renderizarDetalhe(acidente, preencherFormulario);
    if (!acidente) mostrarMensagem(`Registro ${id} não encontrado (HTTP 404).`, "erro");
  } catch (erro) {
    mostrarMensagem(mensagemDeErro(erro), "erro");
  }
}

async function importar(): Promise<void> {
  const arquivo = el<HTMLInputElement>("arquivo").files?.[0];
  if (!arquivo) {
    mostrarMensagem("Selecione um arquivo CSV.", "erro");
    return;
  }
  const botao = el<HTMLButtonElement>("btn-importar");
  botao.disabled = true;
  botao.textContent = "Importando...";
  try {
    const resultado = await api.importar(arquivo);
    renderizarResultado(resultado);
    mostrarMensagem(resultado.mensagem);
    pagina = 0;
    await atualizarTudo();
  } catch (erro) {
    const mensagem = mensagemDeErro(erro);
    mostrarErroImportacao(mensagem);
    mostrarMensagem(mensagem, "erro");
  } finally {
    botao.disabled = false;
    botao.textContent = "Importar";
  }
}

function aplicarFiltros(): void {
  filtro = {
    uf: el<HTMLSelectElement>("filtro-uf").value,
    classificacao: el<HTMLSelectElement>("filtro-classificacao").value,
    dataInicio: el<HTMLInputElement>("filtro-inicio").value,
    dataFim: el<HTMLInputElement>("filtro-fim").value,
  };
  pagina = 0;
  void atualizarTudo();
}

function limparFiltros(): void {
  for (const id of ["filtro-uf", "filtro-classificacao", "filtro-inicio", "filtro-fim"]) {
    el<HTMLInputElement | HTMLSelectElement>(id).value = "";
  }
  filtro = { ...FILTRO_VAZIO };
  pagina = 0;
  void atualizarTudo();
}

function iniciar(): void {

  preencherOpcoes(el<HTMLSelectElement>("campo-classificacao"), CLASSIFICACOES);

  el<HTMLFormElement>("form-acidente").addEventListener("submit", (e) => void salvar(e));
  el("btn-cancelar").addEventListener("click", limparFormulario);
  el("btn-filtrar").addEventListener("click", aplicarFiltros);
  el("btn-limpar-filtros").addEventListener("click", limparFiltros);
  el("btn-buscar").addEventListener("click", () => void buscarPorId());
  el("btn-importar").addEventListener("click", () => void importar());
  el("pag-anterior").addEventListener("click", () => {
    pagina--;
    void carregarTabela();
  });
  el("pag-proxima").addEventListener("click", () => {
    pagina++;
    void carregarTabela();
  });

  void atualizarTudo();
}

iniciar();
```
